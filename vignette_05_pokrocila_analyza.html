Author

Antigravity

Úvod

Tato vigneta demonstruje pokročilou analýzu eyetrackingových dat s využitím balíku eyetools. Místo generovaných dat budeme pracovat s reálným datasetem HCL, který je součástí balíku.

Cílem je ukázat: 1. Práci s reálnými daty (načtení, předzpracování) 2. Využití funkcí eyetools pro vizualizaci a analýzu 3. Pokročilou analýzu přechodů (Transition Matrices)

Výpočet matice přechodů (Transition Matrix)

Spočítáme pravděpodobnost přechodu z AOI A do AOI B.

# Funkce pro přiřazení AOI fixacím (pro účely matice)
assign_aoi <- function(x, y, aois, names) {
  assigned <- "Other"
  for(k in 1:nrow(aois)) {
    # Předpoklad: aois[k,] = c(x, y, w, h) - obdélník se středem x,y
    ax <- aois[k, 1]
    ay <- aois[k, 2]
    aw <- aois[k, 3]
    ah <- aois[k, 4]
    
    if(x >= (ax - aw/2) && x <= (ax + aw/2) &&
       y >= (ay - ah/2) && y <= (ay + ah/2)) {
      assigned <- names[k]
      break
    }
  }
  return(assigned)
}

# Přiřadíme AOI každé fixaci
fixations_df$AOI <- mapply(assign_aoi, fixations_df$x, fixations_df$y, 
                           MoreArgs = list(aois = AOI_areas, names = aoi_names))

# Nyní spočítáme přechody
transitions <- fixations_df %>%
  group_by(pID, trial) %>%
  mutate(
    from = AOI,
    to = lead(AOI)
  ) %>%
  filter(!is.na(to) & from != to) %>% # Zajímají nás jen změny AOI
  ungroup()

# Vytvoření matice
trans_matrix <- transitions %>%
  count(from, to) %>%
  group_by(from) %>%
  mutate(prob = n / sum(n)) %>%
  ungroup()

# Vizualizace Transition Matrix
ggplot(trans_matrix, aes(x = to, y = from, fill = prob)) +
  geom_tile() +
  geom_text(aes(label = round(prob, 2)), color = "white") +
  scale_fill_viridis_c(name = "Pravděpodobnost") +
  theme_minimal() +
  labs(
    title = "Transition Matrix",
    subtitle = "Pravděpodobnost přechodu mezi AOI",
    x = "Do AOI",
    y = "Z AOI"
  )

6. Statistické porovnání

Můžeme porovnat metriky mezi účastníky nebo trialy.

# Agregace dat
stats_df <- aoi_time_stats %>%
  group_by(pID, AOI) %>%
  summarise(
    mean_time = mean(time, na.rm = TRUE),
  )
`summarise()` has grouped output by 'pID'. You can override using the `.groups`
argument.
stats_df_long <- stats_df %>%
  pivot_wider(names_from = AOI, values_from = mean_time, names_prefix = "mean_time_", id_cols = pID)
# Jednoduchý t-test (Left vs Right)
t_test <- t.test(stats_df_long$mean_time_Left, stats_df_long$mean_time_Right, paired = TRUE)

# Vizualizace rozdílů
stats_df_long %>%
  pivot_longer(cols = c(mean_time_Left, mean_time_Right), names_to = "AOI", values_to = "Time") %>%
  ggplot(aes(x = AOI, y = Time, fill = AOI)) +
  geom_boxplot() +
  geom_jitter(width = 0.1) +
  theme_minimal() +
  labs(title = "Porovnání času v levé a pravé AOI")
